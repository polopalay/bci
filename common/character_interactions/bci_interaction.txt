bci_child = {
	icon = pregnant
	use_diplomatic_range = no
	is_shown = {
		scope:actor = {
			is_ai = no
		}
		scope:recipient = {
			is_adult = yes
			age >= 18
		}
	}
	on_accept = {
		scope:recipient = {
			if = {
				limit = { exists = capital_province }
				save_scope_as = child_location
			}
			else_if = {
				limit = { exists = liege }
				liege.capital_province = { save_scope_as = child_location }
			}
			else = {
				scope:actor.capital_province = { save_scope_as = child_location }
			}
			if = {
				limit = { is_female = yes }
				save_scope_as = mother
				if = {
					limit = { exists = primary_spouse }
					primary_spouse = { save_scope_as = father }
				}
			}
			else = {
				save_scope_as = father
				if = {
					limit = { exists = primary_spouse }
					primary_spouse = { save_scope_as = mother }
				}
			}
			create_character = {
				age = 0
				location = scope:child_location
				mother = scope:mother
				father = scope:father
				dynasty_house = house
				culture = culture
				faith = faith
				gender_female_chance = 50
				save_scope_as = child
			}
		}
	}
	auto_accept = yes
}
