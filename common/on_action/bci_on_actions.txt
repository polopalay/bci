bci_viet_betrothal_on_action = {
	trigger = {
		culture = culture:viet
		is_ai = yes
		is_female = no
		is_adult = no
		age >= 8
		is_betrothed = no
	}
	effect = {
		create_character = {
			template = bci_wife
			age = { 2 6 }
			location = root.location
			culture = root.culture
			faith = root.faith
			gender_female_chance = 100
			save_scope_as = bci_betrothal_target
		}
		create_betrothal = scope:bci_betrothal_target
		if = {
			limit = { is_ruler = yes }
			add_courtier = scope:bci_betrothal_target
		}
		else = {
			liege = { add_courtier = scope:bci_betrothal_target }
		}
	}	
}

bci_viet_wife_on_action = {
	trigger = {
		culture = culture:viet

			location = root.location
			culture = root.culture
			faith = root.faith
			gender_female_chance = 100
			save_scope_as = bci_betrothal_target
		}
		marry = scope:bci_betrothal_target
	}	
}
